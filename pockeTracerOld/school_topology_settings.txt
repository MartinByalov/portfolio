═══════════════════════════════════════════════════════════════════════════════
  ROUTER_MAIN — Cisco 2911
  Роля: Интернет шлюз + NAT/PAT за целия офис
═══════════════════════════════════════════════════════════════════════════════

PockeTracer команди:
  connect Router_Main

enable
configure terminal

  hostname Router_Main

  ! ── WAN интерфейс ────────────────────────────────────────────────────────
  ! Gi0/0 е физическата връзка към доставчика (A1/Vivacom/etc.)
  ! /30 маска = само 2 адреса за използване — нашият и на доставчика

  interface GigabitEthernet0/0
    description WAN_TO_ISP
    ip address 94.139.190.2 255.255.255.252
    ip nat outside
    ! "nat outside" = трафикът от тук идва от интернет
    no shutdown
  exit

  ! ── LAN интерфейс (към Core Switch) ─────────────────────────────────────
  ! Тази мрежа 10.0.0.x/30 се нарича "транзитна" — само за пренос между
  ! рутера и Core Switch. Никой компютър не е в нея.

  interface GigabitEthernet0/1
    description TRANSIT_TO_CORE
    ip address 10.0.0.1 255.255.255.252
    ip nat inside
    ! "nat inside" = компютрите зад това са "вътрешните" устройства
    no shutdown
  exit

  ! ── Default Route ─────────────────────────────────────────────────────────
  ! 0.0.0.0 0.0.0.0 = "всички дестинации, които не знам" → изпрати на ISP
  ! Това е маршрутът за интернет достъп

  ip route 0.0.0.0 0.0.0.0 94.139.190.1

  ! ── Статичен маршрут към вътрешните мрежи ─────────────────────────────────
  ! Рутерът трябва да знае: "за 192.168.x.x — питай Core Switch"
  ! 255.255.0.0 покрива всички VLAN мрежи наведнъж (10, 30, 40)

  ip route 192.168.0.0 255.255.0.0 10.0.0.2

  ! ── NAT/PAT конфигурация ─────────────────────────────────────────────────
  ! ACL 1: кои адреси имат право да ползват интернет (всички вътрешни)
  ! Wildcard 0.0.255.255 = обратното на маска /16 → покрива 192.168.0.0-255.255

  access-list 1 permit 192.168.0.0 0.0.255.255

  ! PAT (Port Address Translation) = много компютри, един публичен IP
  ! "overload" е ключовата дума за PAT (разграничава връзките по порт)

  ip nat inside source list 1 interface GigabitEthernet0/0 overload

end
write memory

! ПРОВЕРКИ (след write memory):
  show running-config
  show ip route
  show ip interface brief
  show arp



═══════════════════════════════════════════════════════════════════════════════
  CORE_SWITCH — Cisco Catalyst 3560-24PS (Layer 3 Switch)
  Роля: Сърцето на мрежата — рутира между VLAN-овете
═══════════════════════════════════════════════════════════════════════════════

PockeTracer команди:
  connect Core_Switch

enable
configure terminal

  hostname Core_Switch

  ! ip routing = включва Layer 3 функционалността на суича
  ! Без това командата суичът е обикновен Layer 2 суич

  ip routing

  ! ── VLAN дефиниции ────────────────────────────────────────────────────────
  ! Само задаваме имена тук — портовете се конфигурират на по-долния суич

  vlan 10
    name ADMIN
  vlan 30
    name STUDENTS
  vlan 40
    name CAMERAS
  exit

  ! ── SVI — Switched Virtual Interfaces ────────────────────────────────────
  ! Всеки VLAN получава "виртуален интерфейс" с IP — това е Default Gateway
  ! за всички устройства в съответния VLAN
  ! Адресът .1 е конвенция — може да е и .254 или друг

  interface Vlan10
    ip address 192.168.10.1 255.255.255.0
    ! /24 = 254 използваеми адреса (192.168.10.1 - 192.168.10.254)
    no shutdown
  exit

  interface Vlan30
    ip address 192.168.30.1 255.255.255.0
    no shutdown
  exit

  interface Vlan40
    ip address 192.168.40.1 255.255.255.0
    no shutdown
  exit

  ! ── Routed Port към Router_Main ──────────────────────────────────────────
  ! "no switchport" = превръща суич порта в рутерен порт (Layer 3)
  ! Необходимо за транзитната мрежа 10.0.0.x

  interface GigabitEthernet0/1
    no switchport
    description ROUTED_TO_ROUTER
    ip address 10.0.0.2 255.255.255.252
    no shutdown
  exit

  ! ── Trunk към Switch_Kabinet ──────────────────────────────────────────────
  ! dot1q = IEEE стандартът за VLAN тагове (802.1Q)
  ! Trunk = по един кабел минават VLAN 30 и 40 едновременно (тагирани)

  interface GigabitEthernet0/2
    description TRUNK_TO_KABINET
    switchport trunk encapsulation dot1q
    switchport mode trunk
    no shutdown
  exit

  interface FastEthernet0/24
    description TRUNK_TO_OFFICE
    switchport trunk encapsulation dot1q
    switchport mode trunk
    no shutdown
  exit

  ! ── Default Route ─────────────────────────────────────────────────────────
  ! Цялото интернет-насочено лвижение → нагоре към Router_Main

  ip route 0.0.0.0 0.0.0.0 10.0.0.1

  ! ── DHCP сървър за ученическата мрежа ────────────────────────────────────
  ! Изключваме .1 до .10 = резервирани за мрежово оборудване (суичове, принтери)
  ! Компютрите ще получат адреси от .11 нагоре

  ip dhcp excluded-address 192.168.30.1 192.168.30.10

  ip dhcp pool STUDENT_POOL
    network 192.168.30.0 255.255.255.0
    ! Целият диапазон на VLAN30, минус excluded адресите
    default-router 192.168.30.1
    ! Казваме на компютрите: "Ако не знаеш — питай .1"
    dns-server 8.8.8.8
    ! Google DNS за разпознаване на домейни (google.com → IP)
  exit

end
write memory

! ПРОВЕРКИ:
  show running-config
  show ip route
  show ip interface brief
  show vlan
  show arp



═══════════════════════════════════════════════════════════════════════════════
  SWITCH_KABINET — Cisco Catalyst 2960 (Layer 2 Switch)
  Роля: Директна връзка с ученическите компютри
═══════════════════════════════════════════════════════════════════════════════

PockeTracer команди:
  connect Switch_Kabinet

enable
configure terminal

  hostname Switch_Kabinet

  ! ── VLAN дефиниция ────────────────────────────────────────────────────────
  ! Задължително да дефинираме VLAN и тук, иначе суичът не го знае

  vlan 10
    name ADMIN
  vlan 30
    name STUDENTS
  vlan 40
    name CAMERAS
  exit

  ! ── Access портове за компютри ────────────────────────────────────────────
  ! interface range = конфигурираме 24 порта наведнъж
  ! access mode = "краен порт" — само ЕДИН VLAN минава, без тагове
  ! switchport access vlan 30 = всички компютри са в STUDENT мрежата
  !
  ! spanning-tree portfast = пропуска STP (Spanning Tree) конвергенцията
  ! Нормално STP чака ~30 секунди преди портът да стане активен
  ! PortFast го прави моментално — ОК само за крайни устройства (НЕ за суичове!)

  interface range FastEthernet0/1-24
    switchport mode access
    switchport access vlan 30
    spanning-tree portfast
    no shutdown
  exit

  ! ── Uplink trunk към Core Switch ─────────────────────────────────────────
  ! Gi0/1 е uplink — по него минава трафик от VLAN 30 (и евентуално 40)
  ! trunk mode = много VLAN-ове по един кабел (с 802.1Q тагове)

  interface GigabitEthernet0/1
    description UPLINK_TO_CORE
    switchport mode trunk
    no shutdown
  exit

  ! AP порт (пример: Fa0/8) като студентски достъп
  interface FastEthernet0/8
    description AP_Classrooms
    switchport mode access
    switchport access vlan 30
    spanning-tree portfast
    no shutdown
  exit

end
write memory

  show running-config
  show vlan
  show interfaces

  РЕД НА КОНФИГУРАЦИЯ (ВАЖНО!)
═══════════════════════════════════════════════════════════════════════════════

  1. Постави и свържи всички устройства в canvas
     - Cloud/ISP → Router 2911 (Straight-Through)
     - Router 2911 → Core Switch 3560-24PS (Straight-Through)
     - Core Switch 3560-24PS → Switch_Kabinet (Gi0/2 trunk) (Straight-Through)
     - Core Switch 3560-24PS → Office_Switch (Fa0/24 trunk) (Straight-Through)
     - Switch_Kabinet → PC x24 (Fa0/1-24, access vlan 30) (Straight-Through)
     - Office_Switch → Camera_Server (Fa0/1, vlan 40) + Cameras (Fa0/2-5, vlan 40)
     - Switch_Kabinet → AP_Classrooms (Fa0/8, vlan 30) (Straight-Through)

  2. Конфигурирай Router_Main (пълна Cisco конфигурация по-горе)

  3. Конфигурирай Core_Switch (ip routing, VLANs, SVIs, DHCP)

  4. Конфигурирай Switch_Kabinet (VLANs, access ports, trunk uplink)

  5. Конфигурирай PC_01 (ip address / ip default-gateway или ipconfig за DHCP)

  6. Тествай свързаността:
     connect PC_01
     ping 192.168.30.1    → Core Switch ✓
     ping 10.0.0.1        → Router ✓
     ping 8.8.8.8         → Интернет ✓
     traceroute 8.8.8.8   → Пълен маршрут ✓



═══════════════════════════════════════════════════════════════════════════════
  БЪРЗИ КОМАНДИ (PockeTracer Cheat Sheet)
═══════════════════════════════════════════════════════════════════════════════

  help                         → всички команди
  show devices                 → всички устройства с техните IPs
  show connections             → кабели в топологията
  show topology                → тип топология (Star/Bus/Ring/Mesh/Tree)

  connect <Device_Name>        → отваря конзола към устройство
  disconnect                   → затваря сесията

  enable                       → привилегиран режим (Router#)
  configure terminal           → конфиг режим (Router(config)#)
  end                          → излиза до # режим
  exit                         → излиза едно ниво нагоре
  write memory  / wr           → записва конфигурацията

  show running-config          → цялата конфигурация
  show ip route                → routing таблица
  show ip interface brief      → статус на всички интерфейси
  show interfaces Gi0/0        → детайли за конкретен интерфейс
  show vlan                    → VLAN таблица
  show arp                     → ARP таблица (съседни устройства)
  show version                 → информация за устройството

  ping <ip>                    → тест за свързаност (4 пакета)
  traceroute <ip>              → трасиране на маршрута
  tracert <ip>                 → (Windows алтернатива на traceroute)
  ipconfig                     → IP настройки (стил Windows — за PC/Laptop)

  Tab                          → autocomplete на команди
  ↑↓ стрелки                   → история на командите
